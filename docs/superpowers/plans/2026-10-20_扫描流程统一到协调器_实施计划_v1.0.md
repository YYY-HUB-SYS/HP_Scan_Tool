```python
"""
真机扫描验证脚本 — HP Scan Tool
用途：改造前（D1）与改造后（D2）跑同一流程，产出可对照的 report.json。

用法:
    python tools/verify_scan.py --ip <IP> --pages 1 --out .verify_out/D1_platen
    python tools/verify_scan.py --ip <IP> --pages 3 --out .verify_out/D1_adf

注意：会真的让打印机出纸。
"""

import argparse
import io
import json
import os
import sys
import time
from datetime import datetime

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

import cache_manager
from escl_engine import (
    fetch_capabilities, execute_scan, execute_multipage_scan,
    probe_escl, crop_to_a4,
)


def _a4_crop(path, dpi):
    """与 GUI 的 _apply_post_scan_fixups 行为一致"""
    try:
        with open(path, "rb") as f:
            raw = f.read()
        from PIL import Image
        img = Image.open(io.BytesIO(raw))
        cropped = crop_to_a4(img, dpi=dpi)
        if cropped.size != img.size:
            buf = io.BytesIO()
            cropped.convert("RGB").save(buf, format="JPEG")
            with open(path, "wb") as f:
                f.write(buf.getvalue())
    except Exception as e:
        print("  [warn] A4 裁剪失败: %s" % e)


def main():
    ap = argparse.ArgumentParser(description="HP Scan Tool 真机验证")
    ap.add_argument("--ip", required=True)
    ap.add_argument("--pages", type=int, default=1)
    ap.add_argument("--out", required=True)
    ap.add_argument("--resolution", type=int, default=300)
    ap.add_argument("--source", default="Platen", choices=["Platen", "Feeder"])
    args = ap.parse_args()

    os.makedirs(args.out, exist_ok=True)

    print("[1/5] 探测 eSCL ...")
    url = probe_escl(args.ip, timeout=5.0)
    if not url:
        print("ERROR: 无法探测 eSCL 服务")
        return 2
    print("  eSCL = %s" % url)

    from escl_engine import ScannerInfo
    scanner = ScannerInfo(name="verify", ip=args.ip, escl_url=url)
    scanner = fetch_capabilities(scanner, timeout=5.0)
    print("  型号 = %s  ADF = %s" % (scanner.model, scanner.has_adf))

    job_id = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
    cache_manager.register_job(job_id)

    print("[2/5] 扫描中（打印机将出纸）...")
    t0 = time.time()
    cached = []
    ext = "jpg"
    if args.source == "Feeder":
        pages = execute_multipage_scan(
            scanner=scanner, resolution=args.resolution, color_mode="RGB24",
            output_format="jpg", source="Feeder", duplex=False, timeout=300.0)
        for i, (data, page_ext) in enumerate(pages, 1):
            ext = page_ext
            p = cache_manager.write_page(job_id, i, data, page_ext)
            cached.append(p)
            _a4_crop(p, args.resolution)
    else:
        data, ext = execute_scan(
            scanner=scanner, resolution=args.resolution, color_mode="RGB24",
            output_format="jpg", source="Platen", duplex=False, timeout=120.0)
        p = cache_manager.write_page(job_id, 1, data, ext)
        cached.append(p)
        _a4_crop(p, args.resolution)
    elapsed = time.time() - t0
    print("  完成，耗时 %.1fs，%d 页" % (elapsed, len(cached)))

    print("[3/5] 落盘...")
    from PIL import Image
    base = os.path.join(args.out, "HP_Scan_%s" % datetime.now().strftime("%Y%m%d_%H%M%S"))
    saved = []
    for i, cpath in enumerate(cached, 1):
        target = "%s_%03d.%s" % (base, i, ext)
        img = Image.open(cpath)
        img.convert("RGB").save(target, "JPEG", quality=95)
        saved.append(target)

    print("[4/5] 采集元数据...")
    from PIL import Image
    page_meta = []
    for i, cpath in enumerate(cached, 1):
        img = Image.open(cpath)
        page_meta.append({
            "index": i,
            "cache_file": os.path.basename(cpath),
            "width": img.width,
            "height": img.height,
            "format": img.format,
        })
    for i, spath in enumerate(saved, 1):
        page_meta[i - 1]["saved_file"] = os.path.basename(spath)
        page_meta[i - 1]["saved_bytes"] = os.path.getsize(spath)

    print("[5/5] 清理缓存并写报告...")
    cache_manager.remove_job(job_id)

    report = {
        "device": {"ip": args.ip, "model": scanner.model, "escl_url": url,
                   "has_adf": scanner.has_adf},
        "params": {"resolution": args.resolution, "source": args.source,
                   "requested_pages": args.pages},
        "result": {"page_count": len(cached), "ext": ext,
                   "elapsed_sec": round(elapsed, 1)},
        "pages": page_meta,
    }
    report_path = os.path.join(args.out, "report.json")
    with open(report_path, "w", encoding="utf-8") as f:
        json.dump(report, f, ensure_ascii=False, indent=2)
    print("报告: %s" % report_path)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""
保存落盘管理器 — HP Scan Tool v4.0
把扫描结果写入目标目录：格式转换 / 分组命名 / 额外目录复制 / 缓存清理 / 历史写入。
纯 Python 实现，不依赖 Qt 与任何 GUI，可独立单元测试。
"""

import logging
import os
import shutil

from PIL import Image

import cache_manager
import history_manager

logger = logging.getLogger(__name__)

__all__ = ["build_output_name", "copy_to_extra_dirs", "save_pages", "save_single"]


def build_output_name(base: str, ext: str, group_count: int,
                      group_index: int, in_group_seq: int) -> str:
    """
    生成输出文件名。

    单组（group_count <= 1）—— 保持既有命名不变：
        PDF  -> {base}.pdf
        其他 -> {base}_{组内序号:03d}.{ext}
    多组（group_count > 1）—— 组间不得覆盖：
        PDF  -> {base}_{组号:03d}.pdf
        其他 -> {base}_{组号:03d}_{组内序号:03d}.{ext}
    """
    if ext == "pdf":
        if group_count <= 1:
            return f"{base}.pdf"
        return f"{base}_{group_index:03d}.pdf"
    if group_count <= 1:
        return f"{base}_{in_group_seq:03d}.{ext}"
    return f"{base}_{group_index:03d}_{in_group_seq:03d}.{ext}"


def _save_a4_pdf(img, path, dpi=300):
    """把图片居中放到 A4 画布上，保存为标准尺寸 PDF"""
    a4_w = int(210 / 25.4 * dpi)
    a4_h = int(297 / 25.4 * dpi)
    w, h = img.size
    if w != a4_w or h != a4_h:
        canvas = Image.new("RGB", (a4_w, a4_h), "white")
        x = (a4_w - w) // 2
        y = (a4_h - h) // 2
        if img.mode != "RGB":
            img = img.convert("RGB")
        canvas.paste(img, (x, y))
        canvas.save(path, "PDF", resolution=dpi)
    else:
        img.convert("RGB").save(path, "PDF", resolution=dpi)


def _write_image(img, path, ext, dpi=300):
    """按扩展名落盘一张图片"""
    if ext == "pdf":
        _save_a4_pdf(img, path, dpi=dpi)
    elif ext in ("jpg", "jpeg"):
        img.convert("RGB").save(path, "JPEG", quality=95)
    elif ext == "png":
        img.save(path, "PNG")
    elif ext in ("tiff", "tif"):
        img.save(path, "TIFF")
    else:
        img.convert("RGB").save(path, "JPEG", quality=95)


def copy_to_extra_dirs(saved_files, extra_dirs) -> None:
    """复制到额外输出目录。单个目录失败不影响主产物。"""
    for extra_dir in extra_dirs or []:
        try:
            os.makedirs(extra_dir, exist_ok=True)
            for fpath in saved_files:
                extra_path = os.path.join(extra_dir, os.path.basename(fpath))
                if os.path.abspath(extra_path) != os.path.abspath(fpath):
                    shutil.copy2(fpath, extra_path)
        except Exception as e:
            logger.debug("复制到额外目录失败 [%s]: %s", extra_dir, e)


def _finalize(saved_files, job_id, extra_dirs, device_name, device_ip,
              page_count, ext):
    """额外目录复制 + 清缓存 + 写历史"""
    copy_to_extra_dirs(saved_files, extra_dirs)
    if job_id:
        cache_manager.remove_job(job_id)
    if saved_files:
        history_manager.append({
            "device_name": device_name,
            "device_ip": device_ip,
            "page_count": page_count,
            "format": ext,
            "file_path": saved_files[0],
            "file_count": len(saved_files),
        })
    return saved_files


def save_single(cache_path: str, ext: str, output_path: str, rotation: int = 0,
                extra_dirs=None, job_id: str = None,
                device_name: str = "", device_ip: str = "") -> str:
    """
    单页保存。返回实际落盘路径。

    一律用 with 打开图片：显式管理文件句柄，不依赖 Pillow 内部
    「读取后自动关句柄」的实现细节（该细节随版本变化）。
    """
    with Image.open(cache_path) as src:
        img = src.rotate(rotation, expand=True) if rotation else src.copy()
        final = output_path
        if not final.lower().endswith(f".{ext}"):
            final = f"{final.rsplit('.', 1)[0]}.{ext}"
        _write_image(img, final, ext)
        img.close()
    _finalize([final], job_id, extra_dirs, device_name, device_ip, 1, ext)
    return final


def save_pages(job_id: str, ext: str, output_path: str,
               page_specs, groups, extra_dirs=None,
               device_name: str = "", device_ip: str = "") -> list:
    """
    多页保存（可分组）。

    page_specs: [(缓存文件路径, 旋转角度), ...]，按页序
    groups:     对 page_specs 的索引分组，list[list[int]]
    返回实际落盘的文件路径列表
    """
    base = output_path.rsplit(".", 1)[0]
    group_count = len(groups)
    saved_files = []

    for gi, group_indices in enumerate(groups, 1):
        if ext == "pdf":
            images = []
            for page_idx in group_indices:
                cache_file, rotation = page_specs[page_idx]
                try:
                    with Image.open(cache_file) as src:
                        img = src.rotate(rotation, expand=True) if rotation else src.copy()
                    images.append(img.convert("RGB"))
                except Exception as e:
                    logger.debug("读取缓存页失败 [%s]: %s", cache_file, e)
            if not images:
                continue
            pdf_path = build_output_name(base, ext, group_count, gi, 1)
            images[0].save(pdf_path, "PDF", save_all=True,
                           append_images=images[1:])
            saved_files.append(pdf_path)
        else:
            for seq, page_idx in enumerate(group_indices, 1):
                cache_file, rotation = page_specs[page_idx]
                try:
                    fpath = build_output_name(base, ext, group_count, gi, seq)
                    if ext in ("jpg", "jpeg") and not rotation:
                        shutil.copy2(cache_file, fpath)
                    else:
                        with Image.open(cache_file) as src:
                            img = src.rotate(rotation, expand=True) if rotation else src.copy()
                        _write_image(img, fpath, ext)
                        img.close()
                    saved_files.append(fpath)
                except Exception as e:
                    logger.debug("保存页失败 [%s]: %s", cache_file, e)

    _finalize(saved_files, job_id, extra_dirs, device_name, device_ip,
              len(page_specs), ext)
    return saved_files
```

```python
    def resolve_source(self, scanner, requested: str) -> tuple:
        """
        决定实际扫描来源，返回 (实际来源, 是否探测到输稿器有纸)。

        requested 为协议值："Platen" / "Feeder" / "Auto"
        - "Platen"：用户显式选了平板 → 保持 Platen，**不探测也不覆盖**
        - "Feeder"：用户选了输稿器 → 探测到有纸才用 Feeder，否则回退 Platen
        - "Auto"：探测决定，有纸用 Feeder 否则 Platen
        设备没有 ADF 能力或没有 eSCL 地址时，任何请求都落到 Platen。
        """
        has_adf = getattr(scanner, "has_adf", False)
        if requested == "Platen":
            # 用户显式选择平板：不探测、不覆盖。改造前的 GUI 也是
            # "选平板就是平板"，这里必须保持一致，否则是行为回归。
            return "Platen", False
        if not has_adf or not getattr(scanner, "escl_url", ""):
            return "Platen", False
        try:
            st = get_scanner_status(scanner.escl_url, timeout=5.0)
            loaded = bool(st.get("adf_loaded"))
        except Exception as e:
            logger.debug("ADF 状态检测失败，保持原来源: %s", e)
            if requested == "Feeder":
                return "Feeder", False
            return "Platen", False
        return ("Feeder", True) if loaded else ("Platen", False)

    def run_scan_sync(self, scanner, params: dict, cancel_event=None,
                      job_id: str = None) -> ScanResult:
        """
        同步执行一次扫描并写入缓存。线程池任务与单元测试都走这个方法。
        """
        resolution = params.get("resolution", 300)
        color_mode_ui = params.get("color_mode_ui", "彩色")
        output_format = params.get("output_format", "jpg")
        duplex = params.get("duplex", False)
        output_dir = params.get("output_dir") or DEFAULT_OUT_DIR
        requested = self._source_to_protocol(params.get("source", "平板"))

        source, adf_detected = self.resolve_source(scanner, requested)
        color_mode = COLOR_MODE_MAP.get(color_mode_ui, "RGB24")

        if job_id is None:
            job_id = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
        cache_manager.register_job(job_id)

        cached_paths = []
        ext = output_format

        if source == "Feeder":
            def _progress(n, _total):
                self.scan_progress.emit(job_id, f"已扫描 {n} 页", n)

            pages = execute_multipage_scan(
                scanner=scanner, resolution=resolution, color_mode=color_mode,
                output_format=output_format, source=source, duplex=duplex,
                timeout=300.0, progress_callback=_progress,
                cancel_event=cancel_event,
            )
            for i, (data, page_ext) in enumerate(pages, 1):
                ext = page_ext
                path = cache_manager.write_page(job_id, i, data, page_ext)
                cached_paths.append(path)
                self._apply_a4_crop(path, resolution)
        else:
            data, ext = execute_scan(
                scanner=scanner, resolution=resolution, color_mode=color_mode,
                output_format=output_format, source=source, duplex=duplex,
                timeout=90.0, cancel_event=cancel_event,
            )
            if not data:
                raise ScanCancelled(job_id)
            path = cache_manager.write_page(job_id, 1, data, ext)
            cached_paths.append(path)
            self._apply_a4_crop(path, resolution)

        output_path = os.path.join(
            output_dir,
            f"HP_Scan_{datetime.now().strftime('%Y%m%d_%H%M%S')}.{ext}")

        return ScanResult(
            job_id=job_id, ext=ext, page_count=len(cached_paths),
            cached_paths=cached_paths, output_path=output_path,
            source=source, adf_detected=adf_detected,
        )

    def start_scan(self, scanner, params: dict) -> str:
        """异步启动扫描，立即返回 job_id。结果通过信号通知。"""
        job_id = datetime.now().strftime("%Y%m%d_%H%M%S_%f")
        cancel_event = threading.Event()
        with QMutexLocker(self._cancel_mutex):
            self._cancel_events[job_id] = cancel_event
        self._pool.start(_ScanTask(self, scanner, params, job_id, cancel_event))
        return job_id

    def cancel_scan(self, job_id: str) -> None:
        """取消指定扫描任务"""
        with QMutexLocker(self._cancel_mutex):
            ev = self._cancel_events.get(job_id)
        if ev is not None:
            ev.set()

    def shutdown(self) -> None:
        """取消全部在途任务并停止线程池。GUI 关闭时调用。"""
        with QMutexLocker(self._cancel_mutex):
            events = list(self._cancel_events.values())
        for ev in events:
            ev.set()
        self._pool.clear()
        self._pool.waitForDone(3000)

    @staticmethod
    def _source_to_protocol(source_text: str) -> str:
        """UI 文本 → 协议来源值（Auto 表示自动探测）"""
        if source_text == "自动":
            return "Auto"
        if source_text in SOURCE_MAP:
            return SOURCE_MAP[source_text]
        if source_text in ("Platen", "Feeder", "Auto"):
            return source_text
        return "Platen"

    @staticmethod
    def _apply_a4_crop(cache_path: str, dpi: int) -> None:
        """扫描后处理：A4 居中裁剪（不超过原图）。失败不影响扫描结果。"""
        try:
            from escl_engine import crop_to_a4
            with open(cache_path, "rb") as f:
                raw = f.read()
            img = Image.open(io.BytesIO(raw))
            cropped = crop_to_a4(img, dpi=dpi)
            if cropped.size != img.size:
                buf = io.BytesIO()
                cropped.convert("RGB").save(buf, format="JPEG")
                with open(cache_path, "wb") as f:
                    f.write(buf.getvalue())
        except Exception as e:
            logger.debug("A4 裁剪失败: %s", e)
```

```python
class _ScanTask(QRunnable):
    """在协调器线程池中执行的扫描任务"""

    def __init__(self, coordinator, scanner, params, job_id, cancel_event):
        super().__init__()
        self.setAutoDelete(True)
        self._coordinator = coordinator
        self._scanner = scanner
        self._params = params
        self._job_id = job_id
        self._cancel_event = cancel_event

    def run(self):
        coord = self._coordinator
        job_id = self._job_id
        try:
            coord.scan_started.emit(job_id)
            result = coord.run_scan_sync(self._scanner, self._params,
                                         cancel_event=self._cancel_event,
                                         job_id=job_id)
            if self._cancel_event.is_set():
                coord.scan_cancelled.emit(job_id)
            else:
                coord.scan_finished.emit(job_id, result)
        except ScanCancelled:
            coord.scan_cancelled.emit(job_id)
        except Exception as e:
            logger.error("扫描失败 [%s]: %s", job_id, e, exc_info=True)
            coord.scan_failed.emit(job_id, str(e))
        finally:
            coord._clear_cancel_event(job_id)
```

```python
        with QMutexLocker(self._cancel_mutex):
            self._cancel_events.pop(job_id, None)
```

### 缺陷 1：机型解析失败（真机证据）

2026-10-20 对 17.17.171.111 实测，`GET /eSCL/ScannerCapabilities` 返回：

```xml
<pwg:MakeAndModel>HP LaserJet MFP M232dwc</pwg:MakeAndModel>
<pwg:SerialNumber>VNP6N40672</pwg:SerialNumber>
```

而 `fetch_capabilities` 只匹配 `Model` 标签 ⇒ `scanner.model` 为空串。实测确认：`fetch_capabilities` 返回 `model=''`、`serial='VNP6N40672'`。

影响面（已用 grep 逐点核实）：

| 位置 | 后果 |
|------|------|
| `escl_engine.py:115` `display_name` | 回退成 `WSD-Scanner-17.17.171.111` 这类内部名，GUI 上看不到真实机型 |
| `scan_coordinator.py:55-56` `_ip_label_for` | 设备信息行只显示 IP，缺型号 |
| `scan_coordinator.py:224` | `scan_config.json` 的 `saved_ips[].model` 存空串 |
| `hp_scan_gui_pyside.py:1716/1794/1840/1850` | 扫描历史的 `device_name` 存空 |
| `cli.py:77/128` | 退回 `CLI-<ip>` / 内部名 |

mDNS 发现路径的型号来自 zeroconf 的 `ty` 属性，不受影响；**WSD 发现、手动添加打印机、CLI 这三条路径全靠 `fetch_capabilities` 补全型号，现在全都拿不到**。

### 缺陷 2：ScanJob 409 冲突无法自愈（真机证据）

2026-10-20 对 17.17.171.208 实测，ADF 扫描报 `创建扫描任务失败 HTTP 409`。409 = 打印机上残留一个未完成的 ScanJob（此前某次扫描异常中断留下的）。

实测该型号的能力边界：

| 请求 | 结果 |
|------|------|
| `GET /eSCL/ScanJobs` | 405（不支持列举） |
| `GET /eSCL/ScanJobs/` | 404 |
| `DELETE /eSCL/ScanJobs` | 405（不支持整体删除） |
| `DELETE /eSCL/ScanJobs/` | 404 |

即：**客户端无法主动清理残留 job**，只能等打印机侧 TTL 回收（实测约 15-20 分钟后 409 自动解除）。

**2026-10-20 补充实测（必须与 409 区分开）**：对三台设备分别用 `POST /eSCL/ScanJobs` 建 job（Platen 与 Feeder 两种 InputSource）实测：

| 设备 | AdfState | Platen 建 job | Feeder 建 job |
|------|----------|---------------|---------------|
| 17.17.171.208 | ScannerAdfEmpty | **201 成功** | **503** |
| 17.17.171.111 | - | **201 成功** | **503** |
| 17.17.171.101 | ScannerAdfEmpty | Read timeout | Read timeout |

结论：**输稿器空载时打印机在创建任务阶段就返回 503**（设备在等纸），与 409（残留 job）是两回事。
两者可由状态码区分，但当前代码对 503 只会抛一句『创建扫描任务失败 HTTP 503』，用户看不懂。
注意：3 次短退避（2/4/8 秒，共 14 秒）远短于 TTL（15-20 分钟），所以 409 重试耗尽后必须明确告诉用户『需要等几分钟』，而不是让用户对着一个干巴巴的状态码猜。

当前 `escl_engine.py` 的 `execute_scan` / `execute_multipage_scan` 在 `finally` 里 `session.delete(job_uri)`，能覆盖正常路径，但**覆盖不了进程被杀/断网导致的残留**，也无法应对"打印机上已有别人的残留 job"这种情况。现在的表现是直接抛 `RuntimeError`，用户只能干等且不知道在等什么。

修法：`_create_scan_job` 收到 409 时做有限退避重试（2s / 4s / 8s，共 3 次），重试耗尽则抛出**带用户可执行动作**的错误信息。

### Task 5 fix round 1（控制方 2026-10-20 追加）：301 跳 HTTPS 的老机型

Task 5 的实现修好了 `MakeAndModel` 解析，但**只对 HTTP 直连 200 的机型生效**。实机复验：

| 设备 | ScannerCapabilities | 修复后 `scanner.model` |
|------|--------------------|----------------------|
| 17.17.171.111 | HTTP 200 | `HP LaserJet MFP M232dwc` ✅ |
| 17.17.171.203 | **301** | `""` ❌ |
| 17.17.171.208 | **301** | `""` ❌ |

对 .203/.208，HTTP 被 301 到 HTTPS，而默认 TLS 握手失败：`SSLError: [SSL: SSLV3_ALERT_HANDSHAKE_FAILURE]`。**根因不是证书**（`verify=False` 本来就设了），而是 **OpenSSL 3.x 默认 cipher 套件与老打印机的 TLS 不兼容**。实测：裸 socket 放宽 cipher 后 TLSv1.2 握手成功，urllib3 直连拿到 200 与完整 XML。

**必须修的理由**：.208 能力查询失败 ⇒ 走回退路径 ⇒ `max_width` 取**默认值 2480** 而非设备真值，直接影响 D2 回归的页宽判定基准。顺带解决 HANDOVER 里"HTTPS 301 重定向暂不跟随"这条已知限制。

**顺带发现的既有无效代码**：`fetch_capabilities` 现有的 301 回退 `alt_url = url.replace("https://", "http://")`，而 `url` 实测就是 `http://...`，该 replace **是空操作**，等于又请求了一次同一个 URL。这段逻辑一并重写。

**实现与测试的完整代码**见补充需求文档：
`.superpowers/sdd/2026-10-20_扫描流程统一到协调器_实施计划_v1.0/task-5-fix1-requirements.md`
（新增 `_http_get_loose_tls` 辅助函数；重写 `fetch_capabilities` 的请求段；补 3 条测试：301 指向 https 走宽松 TLS、https 直连失败走宽松 TLS、宽松 TLS 也失败时静默回退）

**安全边界**：仅用于**局域网内自签证书**设备，且只在 HTTP/直连已失败后尝试；`check_hostname=False` + `verify_mode=CERT_NONE` + 宽松 cipher 集合。不放宽到任何互联网访问。
